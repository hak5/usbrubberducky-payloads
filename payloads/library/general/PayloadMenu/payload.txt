VAR $pNUM = FALSE

REM Counters used by CAPS_OFF()
VAR $capTry = 0
VAR $capWait = 0

REM ====================== HELPERS ======================

REM Force CAPS LOCK OFF and VERIFY it, so typed text keeps correct case.
REM Presses CAPSLOCK, then waits (up to 500ms) for the host to confirm
REM the LED is off. Retries up to 3 times if the host never confirms.
REM Only presses again after a full wait, so it can't toggle CAPS back on.
FUNCTION CAPS_OFF()
    $capTry = 0
    WHILE (($_CAPSLOCK_ON == TRUE) && ($capTry < 3))
        CAPSLOCK
        $capWait = 0
        WHILE (($_CAPSLOCK_ON == TRUE) && ($capWait < 10))
            DELAY 50
            $capWait = ($capWait + 1)
        END_WHILE
        $capTry = ($capTry + 1)
    END_WHILE
    DELAY 100
END_FUNCTION

REM Read the current lock states into the "previous" vars
FUNCTION SYNC_LOCKS()
    $pNUM = $_NUMLOCK_ON
    $pSCR = $_SCROLLLOCK_ON
    $pCAP = $_CAPSLOCK_ON
END_FUNCTION

EXTENSION DETECT_READY
    REM VERSION 1.1
    REM AUTHOR: Korben

    REM_BLOCK DOCUMENTATION
        USAGE:
            Extension runs inline (here)
            Place at beginning of payload (besides ATTACKMODE) to act as dynamic
            boot delay

        TARGETS:
            Any system that reflects CAPSLOCK will detect minimum required delay
            Any system that does not reflect CAPSLOCK will hit the max delay of 3000ms
    END_REM

    REM CONFIGURATION:
    DEFINE #RESPONSE_DELAY 25
    DEFINE #ITERATION_LIMIT 120

    VAR $C = 0
    WHILE (($_CAPSLOCK_ON == FALSE) && ($C < #ITERATION_LIMIT))
        CAPSLOCK
        DELAY #RESPONSE_DELAY
        $C = ($C + 1)
    END_WHILE
    CAPSLOCK
END_EXTENSION

REM ====================== STARTUP ======================
DELAY 1000
SAVE_HOST_KEYBOARD_LOCK_STATE

REM Open Notepad via the Run dialog
GUI r
DELAY 700
STRING notepad
ENTER
DELAY 1500
REM Fresh blank doc/tab so CTRL A never wipes a restored Win11 tab
CTRL n
DELAY 800

CAPS_OFF()
DRAW_MENU()
DELAY 200
SYNC_LOCKS()

REM ====================== MAIN LOOP ======================
WHILE (TRUE)
    DELAY 50

    IF ($_NUMLOCK_ON != $pNUM) THEN
        REM NUM LOCK changed -> DOWN (wrap to top)
        $pNUM = $_NUMLOCK_ON
        IF ($sel < #MAXSEL) THEN
            $sel = ($sel + 1)
        ELSE
            $sel = 0
        END_IF
        DRAW_MENU()

    ELSE IF ($_SCROLLLOCK_ON != $pSCR) THEN
        REM SCROLL LOCK changed -> UP (wrap to bottom)
        $pSCR = $_SCROLLLOCK_ON
        IF ($sel > 0) THEN
            $sel = ($sel - 1)
        ELSE
            $sel = #MAXSEL
        END_IF
        DRAW_MENU()

    ELSE IF ($_CAPSLOCK_ON != $pCAP) THEN
        REM CAPS LOCK changed -> SELECT
        CAPS_OFF()
        DO_SELECT()
        REM Let the host's LED reports settle before re-reading states
        DELAY 300
        SYNC_LOCKS()
        DRAW_MENU()
    END_IF
END_WHILE
